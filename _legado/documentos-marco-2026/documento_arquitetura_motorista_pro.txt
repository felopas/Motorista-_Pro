// Custo Fixo Diário
export const calcularCustoFixoDiario = (
  custosFixos: FixedCost[],
  ano: number,
  mes: number
): number => {
  const totalMensal = custosFixos
    .filter(c => c.ativo)
    .reduce((sum, c) => sum + c.valor_mensal, 0);

  const diasUteis = getDiasUteis(ano, mes);
  return totalMensal / diasUteis;
};

// Lucro Líquido
export const calcularLucroLiquido = (
  bruto: number,
  custoFixoDiario: number,
  custosVariaveis: number
): number => {
  return bruto - custoFixoDiario - custosVariaveis;
};

// Meta Diária Dinâmica
export const calcularMetaDiariaAtualizada = (
  metaMensal: number,
  realizado: number,
  diasPlanejados: number,
  diasTrabalhados: number
): { meta: number; urgencia: 'normal' | 'alta' | 'critica' } => {
  const falta = metaMensal - realizado;
  const diasRestantes = diasPlanejados - diasTrabalhados;

  if (diasRestantes <= 0) return { meta: falta, urgencia: 'critica' };

  const metaDiaria = falta / diasRestantes;
  const metaOriginal = metaMensal / diasPlanejados;

  let urgencia: 'normal' | 'alta' | 'critica' = 'normal';
  if (metaDiaria > metaOriginal * 1.3) urgencia = 'critica';
  else if (metaDiaria > metaOriginal * 1.1) urgencia = 'alta';

  return { meta: metaDiaria, urgencia };
};

// Líquido por Hora
export const calcularLiquidoPorHora = (
  liquido: number,
  horas: number
): number => {
  if (horas === 0) return 0;
  return liquido / horas;
};

// Líquido por KM
export const calcularLiquidoPorKM = (
  liquido: number,
  km: number
): number => {
  if (km === 0) return 0;
  return liquido / km;
};

// Percentual da Meta
export const calcularPercentualMeta = (
  realizado: number,
  meta: number
): number => {
  if (meta === 0) return 0;
  return (realizado / meta) * 100;
};

SEÇÃO 6: REGRAS DE EXPERIÊNCIA DO USUÁRIO (UX)
================================================================================

6.1 PRINCÍPIOS NÃO NEGOCIÁVEIS
────────────────────────────────────────────────────────────────────────────────

1. VELOCIDADE: O motorista não pode esperar. Todas as operações devem
   ser síncronas (local-first) e responder em < 100ms.

2. VISIBILIDADE: O lucro líquido deve ser o número mais visível em
   qualquer tela. Não esconda o que importa.

3. URGÊNCIA: A meta diária atualizada deve ser exibida em vermelho
   quando ultrapassar 130% da meta original.

4. SIMPLICIDADE: Máximo 3 toques para registrar um turno completo.

5. FEEDBACK: Cada ação deve ter feedback visual imediato (toast,
   animação, mudança de cor).

6.2 MICRO-INTERAÇÕES
────────────────────────────────────────────────────────────────────────────────

- Botão Registrar: Escala 0.95 no press, volta com spring
- Cards: Hover elevates shadow, border glow
- Gráfico: Animação de preenchimento ao carregar
- Input numérico: Haptic feedback a cada dígito (se disponível)
- Salvar: Confetti ou check animation no sucesso

6.3 ESTADOS DE ERRO
────────────────────────────────────────────────────────────────────────────────

- Sem registro hoje: Card amarelo "Você ainda não registrou hoje"
- Meta impossível: Card vermelho "Meta ajustada: R$ 850/dia"
- Prejuízo: Números em vermelho com ícone de alerta
- Offline: Badge sutil "Modo offline - sincronizando..."
