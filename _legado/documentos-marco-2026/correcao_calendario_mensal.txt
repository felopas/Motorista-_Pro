⚪ BRANCO = Sem registro
🔵 AZUL = Hoje

// Obter dias úteis (excluindo domingos)
function getDiasUteis(ano: number, mes: number): number {
  const totalDias = new Date(ano, mes, 0).getDate();
  let uteis = 0;
  for (let dia = 1; dia <= totalDias; dia++) {
    const data = new Date(ano, mes - 1, dia);
    if (data.getDay() !== 0) uteis++; // 0 = Domingo
  }
  return uteis;
}

// Custo fixo diário do mês (varia por mês!)
function calcularCustoFixoDiario(
  custosTotais: number,
  diasPlanejados: number
): number {
  return custosTotais / diasPlanejados;
}

// Meta diária base (varia por mês!)
function calcularMetaDiaria(
  metaMensal: number,
  diasPlanejados: number
): number {
  return metaMensal / diasPlanejados;
}

// Meta diária dinâmica (recálculo)
function calcularMetaDiariaAtualizada(
  metaMensal: number,
  realizado: number,
  diasPlanejados: number,
  diasTrabalhados: number
): number {
  const falta = metaMensal - realizado;
  const diasRestantes = diasPlanejados - diasTrabalhados;
  return falta / diasRestantes;
}

// Determinar cor do dia no calendário
function getCorDia(registro: DailyRecord, config: MonthConfig): string {
  if (!registro) return '#1e293b'; // Cinza escuro - sem registro
  if (registro.ehFolga) return '#334155'; // Cinza - folga

  const liquido = registro.lucroLiquido;
  const porHora = registro.horas > 0 ? liquido / registro.horas : 0;
  const metaDiaria = config.metaDiaria;

  if (liquido < 0 || porHora < 5) return '#ef4444'; // Vermelho
  if (liquido < metaDiaria || porHora < 7.5) return '#eab308'; // Amarelo
  return '#22c55e'; // Verde
}
